@app.route('/login', methods=['POST'])
def login():
    data = request.get_json() or {}
    usuario = data.get('usuario', '').strip()
    password = data.get('password', '').strip()

    if usuario in USUARIOS and USUARIOS[usuario]['password'] == password:
        user_info = USUARIOS[usuario]
        session['usuario'] = usuario
        session['rol'] = user_info['rol']
        session['equipo'] = user_info['equipo']
        return jsonify({
            "success": True,
            "usuario": usuario,
            "rol": user_info['rol'],
            "equipo": user_info['equipo']
        })
    return jsonify({"success": False, "message": "Usuario o contraseña incorrectos"}), 401

@app.route('/logout', methods=['POST'])
def logout():
    session.clear()
    return jsonify({"success": True})

@app.route('/upload', methods=['POST'])
def upload_file():
    if session.get('rol') != 'lider':
        return jsonify({"success": False, "message": "Se requieren permisos de Líder."}), 403

    if 'file' not in request.files:
        return jsonify({"success": False, "message": "No se adjuntó ningún archivo."}), 400

    file = request.files['file']
    if file.filename == '':
        return jsonify({"success": False, "message": "Nombre de archivo vacío."}), 400

    filename = secure_filename(file.filename)
    filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
    file.save(filepath)

    # ✅ OPCIÓN 1 (Accediendo al índice 1 de la tupla):
    ext = os.path.splitext(filename)[1].lower() 

    try:
        if ext == '.csv':
            try:
                df = pd.read_csv(filepath, encoding='utf-8')
            except Exception:
                df = pd.read_csv(filepath, encoding='latin1', sep=None, engine='python')
        elif ext in ['.xlsx', '.xls']:
            try:
                df = pd.read_excel(filepath, sheet_name='OCTUBRE')
            except Exception:
                df = pd.read_excel(filepath)
        else:
            df = pd.read_csv(filepath, encoding='utf-8')

        return jsonify({
            "success": True,
            "message": f"Archivo '{filename}' subido y procesado con éxito. Total registros: {len(df)}"
        })

    except Exception as e:
        return jsonify({"success": False, "message": f"Error al procesar la planilla: {str(e)}"}), 500
